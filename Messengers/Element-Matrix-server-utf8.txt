


Matrix (Element)

Matrix — это протокол для децентрализованной коммуникации, а Element — клиент для работы с ним. Сервер Matrix называется Synapse.

• Подготовка сервера. Требования: ОС Ubuntu 24.04 LTS, минимум 2 ГБ RAM, 2 ядра CPU, 20 ГБ SSD, статический IP и доступ к интернету.
• Настройка DNS. Обязательна запись A для домена (например, matrix.example.com) и SRV-запись для федерации (_matrix-fed._tcp.matrix.example.com). 
• Обновление системы. sudo apt update && sudo apt upgrade -y
• Установка зависимостей и репозитория Matrix.

sudo apt install -y lsb-release wget apt-transport-https
sudo wget -O /usr/share/keyrings/matrix-org-archive-keyring.gpg https://packages.matrix.org/debian/matrix-org-archive-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/matrix-org-archive-keyring.gpg] https://packages.matrix.org/debian/

$ nano docker-compose.yml

version: '3.8'

networks:
  matrix-network:
    driver: bridge

volumes:
  synapse-data:
    driver: local
    driver_opts:
      type: none
      device: $HOME/Element-Matrix/data/synapse
      o: bind
  postgres-data:
    driver: local
    driver_opts:
      type: none
      device: $HOME/Element-Matrix/data/postgres
      o: bind
  element-data:
    driver: local
    driver_opts:
      type: none
      device: $HOME/Element-Matrix/data/element
      o: bind

services:
  synapse:
    image: matrixdotorg/synapse:latest
    container_name: matrix-synapse
    restart: unless-stopped
    volumes:
      - synapse-data:/data
      - $HOME/Element-Matrix/config/synapse/homeserver.yaml:/data/homeserver.yaml:ro
      - $HOME/Element-Matrix/config/synapse/registration_shared_secret:/data/registration_shared_secret:ro
    environment:
      - SYNAPSE_SERVER_NAME=matrix.yourdomain.com
      - SYNAPSE_REPORT_STATS=no
    networks:
      - matrix-network
    expose:
      - 8008
      - 8448

  postgres:
    image: postgres:14
    container_name: matrix-postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: synapse
      POSTGRES_USER: synapse
      POSTGRES_PASSWORD_FILE: /run/secrets/postgres_password
    secrets:
      - postgres_password
    volumes:
      - postgres-data:/var/lib/postgresql/data
      - $HOME/Element-Matrix/config/postgres/init-db.sh:/docker-entrypoint-initdb.d/init-db.sh
    networks:
      - matrix-network
    expose:
      - 5432

  element:
    image: vectorim/element-web:latest
    container_name: matrix-element
    restart: unless-stopped
    volumes:
      - element-data:/app/data
      - $HOME/Element-Matrix/config/element/config.json:/app/config.json:ro
    networks:
      - matrix-network
    expose:
      - 80

  # Вариант 1: Nginx внутри Docker (рекомендуется для изоляции)
  nginx:
    image: nginx:alpine
    container_name: matrix-nginx
    restart: unless-stopped
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - $HOME/Element-Matrix/config/nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - $HOME/Element-Matrix/config/nginx/certs:/etc/nginx/certs:ro
      - $HOME/Element-Matrix/config/nginx/sites-enabled:/etc/nginx/sites-enabled:ro
      - /etc/letsencrypt:/etc/letsencrypt:ro  # если используете certbot
    networks:
      - matrix-network
    depends_on:
      - synapse
      - element

secrets:
  postgres_password:
    file: $HOME/Element-Matrix/config/postgres/.postgres_password

$ nano config/nginx/sites-enabled/matrix.conf

server {
    listen 80;
    server_name matrix.yourdomain.com;

    # Обработка Matrix-специфичных .well-known запросов (важно для федерации)
    location /.well-known/matrix/ {
        alias /var/www/matrix/;
        try_files $uri $uri/ =404;
    }

    # Редирект всех HTTP-запросов на HTTPS
    location / {
        return 301 https://$host$request_uri;
    }
}

server {
    listen 443 ssl;
    server_name matrix.yourdomain.com;

    # Пути к сертификатам (настройте под свою систему)
    ssl_certificate /etc/nginx/certs/fullchain.pem;
    ssl_certificate_key /etc/nginx/certs/privkey.pem;

    # Основные SSL-настройки
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers on;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;

    # Защита от некоторых атак
    add_header X-Frame-Options DENY;
    add_header X-Content-Type-Options nosniff;
    add_header X-XSS-Protection "1; mode=block";
    add_header Referrer-Policy "no-referrer-when-downgrade";

    # Корневой путь — проксируем на Element
    location / {
        proxy_pass http://element:80;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Real-IP $remote_addr;

        # Настройки буферизации для WebSocket
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 86400;
    }

    # API Matrix (клиентская часть и федерация)
    location /_matrix {
        proxy_pass http://synapse:8008;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Real-IP $remote_addr;

        # Для WebSocket-соединений
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 86400;
    }

    # Альтернативный путь для клиента (если нужен отдельный доступ)
    location /client/ {
        proxy_pass http://element:80/;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Статика Element (иконки, CSS и т.п.)
    location /static/ {
        proxy_pass http://element:80/static/;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # Защита административных эндпоинтов
    location = /_synapse/admin {
        deny all;
    }

    # Логирование
    access_log /var/log/nginx/matrix_access.log;
    error_log /var/log/nginx/matrix_error.log;
}

Настройка Firewalld для Matrix‑сервера (по принципам из статьи)

Исходя из концепции разделения зон (как в вашей статье), оптимально создать отдельную зону для Matrix‑сервисов. Это даст:

• чёткое разграничение трафика;
• лёгкость аудита правил;
• возможность быстро отключить/изменить правила для Matrix без влияния на другие сервисы.

Создание зоны matrix

sudo firewall-cmd --permanent --new-zone=matrix

Назначение интерфейсов (если нужно)

sudo firewall-cmd --permanent --zone=matrix --add-interface=eth1

Открытие портов в зоне matrix

# HTTP/HTTPS для Let's Encrypt и работы Element
sudo firewall-cmd --permanent --zone=matrix --add-service=http
sudo firewall-cmd --permanent --zone=matrix --add-service=https
# Порт федерации Matrix
sudo firewall-cmd --permanent --zone=matrix --add-port=8448/tcp
# Если нужен доступ к Portainer (например, порт 9000)
sudo firewall-cmd --permanent --zone=matrix --add-port=9000/tcp

Привязка IP‑адреса (опционально)

sudo firewall-cmd --permanent --zone=matrix --add-source=203.0.113.45

Для нескольких IP:

sudo firewall-cmd --permanent --zone=matrix --add-source=192.168.1.0/24

Активация зоны и перезагрузка

sudo firewall-cmd --reload

Проверка

# Посмотреть все правила зоны
sudo firewall-cmd --zone=matrix --list-all
# Проверить активные зоны
sudo firewall-cmd --get-active-zones

Итог:

• Все правила для Matrix собраны в зоне matrix.
• Если нужно отключить Matrix — достаточно удалить зону:

sudo firewall-cmd --permanent --delete-zone=matrix
sudo firewall-cmd --reload
